## 1. **Layout & Responsive Design**

### Current Issues:
- Fixed 250x250px size doesn't adapt to content
- Single-column layout for multiple images (wastes space)
- No support for different aspect ratios

### Improvements:

```typescript
// Add responsive grid layout for multiple images
const renderAdditionalData = () => {
  if (!additional_data || additional_data.length === 0) return null;
  
  const images = additional_data.filter(item => item.data_type === 'img');
  const videos = additional_data.filter(item => item.data_type === 'video');
  
  return (
    <View style={styles.additionalDataContainer}>
      {/* Image Grid */}
      {images.length > 0 && (
        <View style={[
          styles.imageGrid,
          images.length === 1 && styles.singleImage,
          images.length === 2 && styles.twoImages,
          images.length >= 3 && styles.multipleImages
        ]}>
          {images.map((item, index) => renderImage(item, index))}
        </View>
      )}
      
      {/* Videos (full width) */}
      {videos.map((item, index) => renderVideo(item, index))}
    </View>
  );
};
```

```typescript
const [imageLoadingStates, setImageLoadingStates] = useState<Record<number, boolean>>({});

const renderImage = (item: AdditionalData, index: number) => {
  const isLoading = imageLoadingStates[index] ?? true;

              source={{ uri: item.data }}

                styles.generatedImage,

              style={[
                styles.generationBadge,

                <Text style={styles.imageCountText}>

                </Text>
              </View>
            )}
          </Pressable>
        </Link>
      </ContextMenu.Trigger>
      <ContextMenu.Content>{/* ... */}</ContextMenu.Content>
    </ContextMenu.Root>
  );
};
```

## 4. **Improved Video Handling**

### Add Download Progress:

```typescript
const [videoDownloadProgress, setVideoDownloadProgress] = useState<Record<number, number>>({});

const downloadVideoWithProgress = async (url: string, index: number) => {
  try {
    setLoadingDownloadVideo(true);
    
    // Use FileSystem to track download progress
    const callback = (downloadProgress: DownloadProgressData) => {
      const progress = downloadProgress.totalBytesWritten / downloadProgress.totalBytesExpectedToWrite;
      setVideoDownloadProgress(prev => ({...prev, [index]: progress}));
    };
    
    const downloadResumable = FileSystem.createDownloadResumable(
      url,
      FileSystem.documentDirectory + 'video.mp4',
      {},
      callback
    );
    
    const result = await downloadResumable.downloadAsync();
    if (result) {
      await MediaLibrary.saveToLibraryAsync(result.uri);
      Alert.alert('Success', 'Video saved to gallery');
    }
  } catch (error) {
    Alert.alert('Error', 'Failed to download video');
  } finally {
    setLoadingDownloadVideo(false);
    setVideoDownloadProgress(prev => ({...prev, [index]: 0}));
  }
};
```

### Display Progress:

```typescript
<TouchableOpacity
  style={styles.downloadVideoButton}
  onPress={() => downloadVideoWithProgress(item.data, index)}
  disabled={isLoadingDownloadVideo}
>

      </Text>

</TouchableOpacity>
```

```typescript
// Only render images when they're visible
import { useInView } from 'react-native-intersection-observer';

const LazyImage = ({ item, index }) => {
  const { ref, inView } = useInView({
    triggerOnce: true,
    threshold: 0.1,
  });
  
  return (
    <View ref={ref} style={styles.imageWrapper}>
      {inView ? (
        <Image source={{ uri: item.data }} style={styles.generatedImage} />
      ) : (
        <View style={[styles.generatedImage, styles.placeholder]} />
      )}
    </View>
  );
};
```

## 6. **Enhanced User Interactions**

### Pinch to Zoom (without leaving chat):

```typescript
import { GestureDetector, Gesture } from 'react-native-gesture-handler';
import Animated, { useSharedValue, useAnimatedStyle } from 'react-native-reanimated';

const ImageWithZoom = ({ uri }) => {
  const scale = useSharedValue(1);
  const savedScale = useSharedValue(1);
  
  const pinchGesture = Gesture.Pinch()
    .onUpdate((e) => {
      scale.value = savedScale.value * e.scale;
    })
    .onEnd(() => {
      savedScale.value = scale.value;
    });
  
  const animatedStyle = useAnimatedStyle(() => ({
    transform: [{ scale: scale.value }],
  }));
  
  return (
    <GestureDetector gesture={pinchGesture}>
      <Animated.Image
        source={{ uri }}
        style={[styles.generatedImage, animatedStyle]}
      />
    </GestureDetector>
  );
};
```
